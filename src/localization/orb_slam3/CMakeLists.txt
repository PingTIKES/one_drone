cmake_minimum_required(VERSION 3.16)
project(orb_slam3)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(cv_bridge REQUIRED)
find_package(OpenCV 4 REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(Boost REQUIRED COMPONENTS serialization)
find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_PANGOLIN_VIDEO OFF CACHE BOOL "" FORCE)
set(BUILD_PANGOLIN_PYTHON OFF CACHE BOOL "" FORCE)
add_subdirectory(vendor/Pangolin)
add_subdirectory(vendor/ORB_SLAM3/Thirdparty/DBoW2)
add_subdirectory(vendor/ORB_SLAM3/Thirdparty/g2o)
file(GLOB ORB_CORE_SOURCES CONFIGURE_DEPENDS vendor/ORB_SLAM3/src/*.cc vendor/ORB_SLAM3/src/*.cpp vendor/ORB_SLAM3/src/CameraModels/*.cpp)
add_library(orb_slam3_core SHARED ${ORB_CORE_SOURCES})
target_include_directories(orb_slam3_core PUBLIC vendor/ORB_SLAM3 vendor/ORB_SLAM3/include vendor/ORB_SLAM3/include/CameraModels vendor/ORB_SLAM3/Thirdparty/Sophus ${OpenCV_INCLUDE_DIRS})
target_link_libraries(orb_slam3_core PUBLIC DBoW2 g2o pangolin Eigen3::Eigen ${OpenCV_LIBS} Boost::serialization OpenSSL::Crypto Threads::Threads)
add_executable(stereo_inertial src/stereo_inertial.cpp)
target_link_libraries(stereo_inertial orb_slam3_core)
ament_target_dependencies(stereo_inertial rclcpp sensor_msgs nav_msgs std_msgs cv_bridge)
# Unpack the vocabulary into build/, never dirty or duplicate it in the source checkout.
if(NOT EXISTS "${CMAKE_CURRENT_BINARY_DIR}/ORBvoc.txt")
  execute_process(COMMAND ${CMAKE_COMMAND} -E tar xzf "${CMAKE_CURRENT_SOURCE_DIR}/vendor/ORB_SLAM3/Vocabulary/ORBvoc.txt.tar.gz" WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}" RESULT_VARIABLE VOC_RESULT)
  if(NOT VOC_RESULT EQUAL 0)
    message(FATAL_ERROR "Cannot extract bundled ORB vocabulary")
  endif()
endif()
install(TARGETS orb_slam3_core DBoW2 g2o LIBRARY DESTINATION lib)
install(TARGETS stereo_inertial DESTINATION lib/${PROJECT_NAME})
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/ORBvoc.txt" DESTINATION share/${PROJECT_NAME}/vocabulary)
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/prepare_calibration.py DESTINATION lib/${PROJECT_NAME})
ament_package()
