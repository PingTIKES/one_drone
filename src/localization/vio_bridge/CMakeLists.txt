cmake_minimum_required(VERSION 3.8)
project(vio_bridge)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
ament_python_install_package(${PROJECT_NAME} PACKAGE_DIR src)
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
set(VIO_BRIDGE_SCRIPT "${CMAKE_CURRENT_BINARY_DIR}/vio_to_px4.py")
configure_file(src/vio_to_px4.py "${VIO_BRIDGE_SCRIPT}" COPYONLY)
file(CHMOD "${VIO_BRIDGE_SCRIPT}" PERMISSIONS OWNER_READ OWNER_WRITE OWNER_EXECUTE GROUP_READ GROUP_EXECUTE WORLD_READ WORLD_EXECUTE)
install(PROGRAMS "${VIO_BRIDGE_SCRIPT}" DESTINATION lib/${PROJECT_NAME})
ament_package()
